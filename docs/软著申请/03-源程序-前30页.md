/**
 * @file App.tsx
 * @description 应用外壳（App Shell）：只负责视图路由、登录态持久化与全局弹窗宿主。
 *
 * 具体业务页面（馆空间）位于 src/ui/space，登录页位于 src/ui/auth，
 * 本文件不承载任何场景陈设、摆件或弹窗内容。
 */
import React, { useEffect, useState } from 'react';
import { AuthPage } from './ui/auth';
import { HallSpacePage } from './ui/space';
import { openAgreement } from './ui/agreement';
import { useViewRouter } from './ui/shell/viewRouter';
import type { UserProfile } from './types';

const USER_STORAGE_KEY = 'memery_current_user';

const readStoredUser = (): UserProfile | null => {
  try {
    const saved = localStorage.getItem(USER_STORAGE_KEY);
    return saved ? (JSON.parse(saved) as UserProfile) : null;
  } catch {
    return null;
  }
};

const App: React.FC = () => {
  const { view, navigate } = useViewRouter();
  const [currentUser, setCurrentUser] = useState<UserProfile | null>(readStoredUser);

  // 登录态持久化集中在外壳，页面不直接接触存储
  useEffect(() => {
    try {
      if (currentUser) {
        localStorage.setItem(USER_STORAGE_KEY, JSON.stringify(currentUser));
      } else {
        localStorage.removeItem(USER_STORAGE_KEY);
      }
    } catch (err) {
      console.error(err);
    }
  }, [currentUser]);

  if (view === 'auth') {
    return (
      <div className="relative w-full min-h-screen bg-[#0c0a09]">
        <AuthPage
          onBack={() => navigate('hall')}
          onSuccess={(user) => {

                                              暖念纪念场景模拟软件 V1.0            第 1 页
───────────────────────────────────────────────────────────────────────────────────

            setCurrentUser(user);
            navigate('hall');
          }}
          onOpenAgreement={openAgreement}
        />
      </div>
    );
  }

  return (
    <HallSpacePage
      currentUser={currentUser}
      onRequestAuth={() => navigate('auth')}
      onUserChange={setCurrentUser}
    />
  );
};

export default App;

/**
 * 把场景物件 id 映射到档案 id。
 * 供瓶优先返回所插花材的典故，其次器型，最后落到「供瓶」总条目。
 */
export const resolveSceneItemLoreId = (sceneId: string, item?: LoreSceneItemHint): string | undefined => {
  if (sceneId === 'censer') return 'censer';
  if (sceneId.startsWith('candle')) return 'candle';
  if (sceneId === 'tea_cups') return 'tea_cups';
  if (sceneId === 'singing_bowl') return 'singing_bowl';

  if (sceneId.startsWith('fruit')) {
    return item?.fruitType || undefined;
  }

  if (sceneId.startsWith('vase') || sceneId.startsWith('offered_flower')) {
    if (item?.flowerType && hasRelicLore(item.flowerType)) return item.flowerType;
    if (item?.vaseStyle && hasRelicLore(item.vaseStyle)) return item.vaseStyle;
    return 'vase';
  }

  return hasRelicLore(sceneId) ? sceneId : undefined;
};

/** 关键词检索（名称、诗意名、朝代、典故、诗句） */
export const searchRelicLore = (keyword: string): RelicLore[] => {
  const kw = keyword.trim().toLowerCase();
  if (!kw) return ALL_LORE;
  return ALL_LORE.filter((item) =>
    [item.name, item.poeticTitle, item.dynasty, item.origin, item.allusion, item.poem?.text, item.poem?.title]
      .filter(Boolean)
      .some((field) => (field as string).toLowerCase().includes(kw))
  );
};
